# cmdlet - bold to code span
\*\*(\w+-\w+)\*\*
`$1`

# cmdlet/vars - bare to code spans
([\n\s])(\w+-\w+)([\n\s\.,])
([\n\s])(\$\w+)([\n\s\.,])
$1`$2`$3

# parameters - change italic to bold
([\n\s])(\*\w+\*)([\n\s\.,])
$1*$2*$3

# italics - change asterisk to underscore
([\n\s])(\*)(\w+)(\*)([\n\s\.,])
$1_$3_$5

# parameters - change code span to bold
(`-)(\w+)(`)
**$2**

# GitHub item links
(\()#(\d+)(\))
$1[#$2][$2]$3

#------ formats -----------
# convert italics to use underscores
([\n\s])(\*)(\w+.*\w+)(\*)([\n\s\.,])
$1_$3_$5

# change bold-underscores to asterisk
__([^_]+)__
**$1**

# add blank line after headers
(#{1,3}\s.+\n)([^\n])
$1\n$2

# change keystrokes to kbd
(Ctrl|Shift|Alt)[-+]([^\s]+)
<kbd>$1</kbd>+<kbd>$2</kbd>

# change bold keystrokes to kbd
(\*\*)(\w+)\s*\+\s*(.+)(\*\*)
<kbd>$2</kbd>+<kbd>$3</kbd>

# undo blank line after headers for SYNOPSIS
(#{1,3}\s[\W*(SYNOPSIS)\W*]+\n)([\n])
$1

# add blank space at the beginning of each line in the syntax block
# Note the single white space in front of $0
^\[-|^\[\[|^\[<
 $0

#------ contractions -----------
# NOT
([\n\s])(\w+)([\n\s]not)([\n\s])
$1$2n't$4

#------ Search patterns -----------
# Find indented code block
```powershell(\n(\s\s\s\s)[^\n]+)+

# Find hyperlinks
\[[^\]]+\]\(

# Find bad syntax reflows (split parameter values)
^<

# Find double line returns
\n\n
\n

# Look for unnecessary line returns i.e. short lines that would render in the same paragraph anyway
(^.{0,80}\.)(\n^[a-zA-Z])

# Look for non-reflowed lines
.{101,}

# compare module guids for manifest vs markdown
dir | %{
  $md = select-string -path "$($_.name)\$($_.name).md" -pattern 'Module Guid:','Module Name:'
  $file = [pscustomobject]@{ Src = 'MD'; Name = ''; Guid = '' }
  $md.line | %{
    $x = $_.split(':')
    if ($x[0] -like '*Guid') {
      $file.Guid = $x[1].trim()
    } else {
      $file.Name = $x[1].trim()
    }
  }
  $mod = gmo $_.name -list | select -first 1 -prop @{n='Src';e={'PS'}},Name,Guid
  compare-object $mod $file -prop name,guid
}

# Regex modes
Modifier group   (?letters:regex) Non-capturing group with modifiers that affect only the part of the regex inside the group.
Explicit capture (?n)             Plain parentheses are non-capturing groups instead of numbered capturing groups.
                                  Only named capturing groups actually captured.
Case insensitive (?i)             Turn on case insensitivity.
Free-spacing     (?x)             Turn on free-spacing mode to ignore whitespace between regex tokens and allow # comments.
Single-line      (?s)             Make the dot match all characters including line break characters.
Multi-line       (?m)             Make ^ and $ match at the start and end of each line.
